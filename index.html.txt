<script>

const totalDays = 30;

let currentDay = 1;

const tasks = {};

for(let i=1;i<=totalDays;i++){

  tasks[i] = [

    {
      name:"運動 30 分鐘",
      points:10,
      done:false
    },

    {
      name:"閱讀",
      points:5,
      done:false
    },

    {
      name:"完成指定工作",
      points:15,
      done:false
    }

  ];

}

// 儲存

function saveData(){

  localStorage.setItem(
    "juneMissionData",
    JSON.stringify(tasks)
  );

}

// 載入

function loadData(){

  const saved =
    localStorage.getItem("juneMissionData");

  if(saved){

    const parsed = JSON.parse(saved);

    Object.assign(tasks, parsed);

  }

}

// 計算單日分數

function getDayScore(day){

  return tasks[day]
    .filter(t=>t.done)
    .reduce((sum,t)=>sum+t.points,0);

}

// 計算總分

function getTotalScore(){

  let total = 0;

  for(let i=1;i<=totalDays;i++){

    total += getDayScore(i);

  }

  return total;

}

// 日曆

function renderCalendar(){

  const calendar =
    document.getElementById("calendar");

  calendar.innerHTML = "";

  for(let day=1;day<=totalDays;day++){

    const score = getDayScore(day);

    const div = document.createElement("div");

    div.className = "day";

    if(day === currentDay){

      div.classList.add("active");

    }

    div.innerHTML = `

      <div class="day-number">
        ${day}
      </div>

      <div class="day-score">
        ${score} 分
      </div>

    `;

    div.onclick = ()=>{

      currentDay = day;

      renderAll();

    };

    calendar.appendChild(div);

  }

}

// 任務

function renderTasks(){

  document.getElementById("currentDate")
    .innerText = `6月${currentDay}日`;

  const taskList =
    document.getElementById("taskList");

  taskList.innerHTML = "";

  tasks[currentDay].forEach((task,index)=>{

    const div = document.createElement("div");

    div.className = "task";

    div.innerHTML = `

      <div class="task-left">

        <input
          type="checkbox"
          ${task.done ? "checked" : ""}
          onchange="toggleTask(${currentDay},${index})"
        >

        <div class="task-name">
          ${task.name}
        </div>

      </div>

      <div class="points">
        +${task.points}
      </div>

    `;

    taskList.appendChild(div);

  });

}

// 切換任務

function toggleTask(day,index){

  tasks[day][index].done =
    !tasks[day][index].done;

  saveData();

  renderAll();

}

// 總結

function renderSummary(){

  const total = getTotalScore();

  document.getElementById("totalScore")
    .innerText = `${total} 分`;

  const maxScore = 200;

  const percent =
    Math.min((total / maxScore) * 100,100);

  document.getElementById("progressBar")
    .style.width = percent + "%";

}

// 切換日期

function changeDay(direction){

  currentDay += direction;

  if(currentDay < 1){

    currentDay = 1;

  }

  if(currentDay > totalDays){

    currentDay = totalDays;

  }

  renderAll();

}

// 更新全部

function renderAll(){

  renderCalendar();

  renderTasks();

  renderSummary();

}

loadData();

renderAll();

</script>
